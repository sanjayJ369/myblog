{{- /* The shallowest heading level used in a post's Markdown (code fences ignored). Posts written with "#" get 1. */}} {{- $md := replaceRE "(?s)```.*?```|~~~.*?~~~" "" .RawContent }} {{- $top := 2 }} {{- if findRE `(?m)^#\s` $md 1 }}{{ $top = 1 }}{{ else if not (findRE `(?m)^##\s` $md 1) }}{{ $top = 3 }}{{ end }} {{- return $top }}
