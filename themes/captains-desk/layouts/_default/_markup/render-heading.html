{{- /* The post's section level becomes an engraved, §-numbered

; deeper levels shift with it. A lone heading above that level (e.g. "# Title" over "###" sections) renders as an unnumbered lead. Built as one string because Go templates escape a dynamic tag name. */}} {{- $top := partial "top-level.html" .Page }} {{- $depth := sub .Level $top }} {{- $n := int (math.Min 6 (math.Max 2 (add $depth 2))) }} {{- $class := cond (lt $depth 0) "lead" (cond (eq $depth 0) "sec" (cond (eq $depth 1) "sub" "subsub")) }} {{- printf `%s` $n (.Anchor | htmlEscape) $class .Text $n | safeHTML }}
